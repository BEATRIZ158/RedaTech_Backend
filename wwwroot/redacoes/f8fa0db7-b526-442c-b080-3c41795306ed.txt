UsuarioModel
public class UsuarioModel
{
    [Key]
    public int Id { get; set; } //Id
    public string Nome { get; set; } //Nome
    public string Email { get; set; } //email
    public string Cpf { get; set; } //CPF
    public bool Status { get; set; } //Ativa/Inativo

    [Column(TypeName = "date")]
    public DateTime DataNascimento { get; set; }
    public DateTime DataDeCriacao { get; set; } = DateTime.Now.ToLocalTime(); //Data da criação
    public TipoUsuario TipoUsuario { get; set; }
}

RedacaoModel
public class RedacaoModel
{
    [Key]
    public int Id { get; set; }
    public string CaminhoArquivo { get; set; } = string.Empty;
    public DateTime DataDeEnvio { get; set; } = DateTime.Now.ToLocalTime();
    public int AlunoId { get; set; }

    //Propriedade de navegação para UsuarioModel
    [ForeignKey("AlunoId")]
    public UsuarioModel Aluno { get; set; }
}

CorrecaoModel
public class CorrecaoModel
{
    [Key]
    public int Id { get; set; }

    [ForeignKey("Redacao")]
    public int RedacaoId { get; set; }
    public RedacaoModel Redacao { get; set; }

    [ForeignKey("Corretor")]
    public int CorretorId { get; set; }
    public UsuarioModel Corretor { get; set; }

    public string Comentarios { get; set; }
    public StatusCorrecao Status { get; set; } = StatusCorrecao.EmAndamento;
    public DateTime DataCorrecao { get; set; } = DateTime.Now;
}

ServiceResponse
